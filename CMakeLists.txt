cmake_minimum_required(VERSION 3.15)
project(histogram_node_study LANGUAGES CXX)
find_package(correctionlib CONFIG REQUIRED)
find_path(RAPIDJSON_INCLUDE_DIR rapidjson/document.h REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_json_numbers src/json_numbers.cc)
target_include_directories(_json_numbers SYSTEM PRIVATE ${RAPIDJSON_INCLUDE_DIR})
target_compile_options(_json_numbers PRIVATE -ffp-contract=off)
install(TARGETS _json_numbers LIBRARY DESTINATION histogram_nodes)
pybind11_add_module(_batch_probe src/batch_probe.cc)
target_link_libraries(_batch_probe PRIVATE correctionlib)
target_compile_options(_batch_probe PRIVATE -O3 -march=native -ffp-contract=off)
set_target_properties(_batch_probe PROPERTIES INSTALL_RPATH "$ORIGIN/../correctionlib/lib")
install(TARGETS _batch_probe LIBRARY DESTINATION histogram_nodes)
file(GLOB NODE_SOURCES CONFIGURE_DEPENDS src/nodes/*.cc)
add_library(histogram_nodes SHARED src/registry.cc src/table.cc ${NODE_SOURCES})
target_compile_features(histogram_nodes PRIVATE cxx_std_17)
target_include_directories(histogram_nodes PRIVATE src)
target_include_directories(histogram_nodes SYSTEM PRIVATE ${RAPIDJSON_INCLUDE_DIR})
target_link_libraries(histogram_nodes PRIVATE correctionlib)
target_compile_options(histogram_nodes PRIVATE -O3 -march=native -ffp-contract=off -Wall -Wextra)
option(STUDY_SANITIZE "Enable undefined behavior and float-cast-overflow checks" OFF)
if(STUDY_SANITIZE)
  target_compile_options(histogram_nodes PRIVATE -fsanitize=undefined,float-cast-overflow -fno-sanitize-recover=all)
  target_link_options(histogram_nodes PRIVATE -fsanitize=undefined,float-cast-overflow)
endif()
set_target_properties(histogram_nodes PROPERTIES INSTALL_RPATH "$ORIGIN/../correctionlib/lib")
install(TARGETS histogram_nodes LIBRARY DESTINATION histogram_nodes)
